Tugas 8 PBKK — Landing Page E-Commerce ShopEase dengan CodeIgniter 3

📂 Source code lengkap: github.com/miaikoo/2026-framework-based-programming

Pada tugas ini saya membuat landing page toko online ShopEase menggunakan CodeIgniter 3.1.13 dengan memodifikasi routing, controller, dan view. ShopEase mengikuti studi kasus jurnal "Website E-Commerce Menggunakan Model View Controller (MVC) dengan Framework CodeIgniter — Studi Kasus: Toko Miniatur" (Jurnal Ilmiah DASI Vol. 16 No. 1, 2015), yaitu toko miniatur di Yogyakarta yang beralih dari penjualan konvensional ke toko online.

Cara Menjalankan

  1. Start Apache di XAMPP (tidak perlu database; data katalog ada di model).
  2. Letakkan folder di htdocs, lalu buka http://localhost/pbkk8/shopease/.

Karena CodeIgniter 3.1.13 dirilis sebelum PHP 8.2, pesan Deprecated di PHP 8.5 disembunyikan lewat error_reporting(E_ALL & ~E_DEPRECATED & ~E_USER_DEPRECATED) di index.php. Konfigurasi lain: base_url otomatis, index_page kosong + .htaccess, CSRF aktif, dan bahasa Indonesia.

1. Modifikasi Routing

// ---------------------------------------------------------------
// Routing ShopEase
// ---------------------------------------------------------------
// Halaman utama (landing page)
$route['default_controller'] = 'home';

// Katalog: /kategori/kereta, /cari?q=vespa, /produk/lokomotif-uap
$route['kategori/(:any)'] = 'home/kategori/$1';
$route['cari']            = 'home/cari';
$route['produk/(:any)']   = 'produk/detail/$1';

// Newsletter (form POST di landing page)
$route['langganan'] = 'home/langganan';

// Keranjang belanja (shopping cart)
$route['keranjang']          = 'keranjang/index';
$route['keranjang/tambah']   = 'keranjang/tambah';
$route['keranjang/hapus']    = 'keranjang/hapus';

// Halaman statis: satu controller untuk beberapa URL
$route['(tentang|cara-belanja|kontak)'] = 'halaman/tampil/$1';

// Halaman 404 buatan sendiri
$route['404_override'] = 'halaman/tidak_ditemukan';
$route['translate_uri_dashes'] = FALSE;
  • /kategori/kereta → Home::kategori('kereta'): wildcard (:any) diteruskan sebagai parameter.
  • /produk/kapal-pinisi → Produk::detail('kapal-pinisi'): URL memakai slug, bukan ID.
  • (tentang|cara-belanja|kontak): satu route regex untuk tiga halaman sekaligus.
  • 404_override: URL yang tidak dikenal menampilkan halaman 404 bergaya ShopEase.

2. Modifikasi Controller

core/MY_Controller.php — data global, layout, dan 404

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * Controller dasar ShopEase.
 * Menyiapkan data yang tampil di setiap halaman (menu kategori, jumlah isi keranjang)
 * dan method render() untuk membungkus view ke dalam layout utama.
 */
class MY_Controller extends CI_Controller
{
    public function __construct()
    {
        parent::__construct();
        $this->load->model('katalog_model');
        $this->load->vars(array(
            'menu_kategori'   => $this->katalog_model->kategori(),
            'jumlah_keranjang' => array_sum((array) $this->session->userdata('keranjang')),
            'layanan'         => $this->katalog_model->layanan(),
        ));
    }

    /**
     * @param string $view  view isi halaman
     * @param array  $data  data untuk view; 'title' dipakai di <title>
     */
    protected function render($view, array $data = array())
    {
        $data['content'] = $this->load->view($view, $data, TRUE);
        $this->load->view('layouts/main', $data);
    }

    /**
     * Tampilkan halaman 404 ShopEase lalu hentikan proses.
     * show_404() bawaan tidak melewati $route['404_override'], jadi dipakai method ini
     * agar produk/kategori yang tidak ada tetap memakai tampilan yang sama.
     */
    protected function tidak_ada()
    {
        $this->output->set_status_header(404);
        $this->render('halaman/tidak_ditemukan', array(
            'title' => 'Halaman tidak ditemukan — ShopEase',
            'saran' => $this->katalog_model->terlaris(3),
        ));
        $this->output->_display();
        exit;
    }
}

controllers/Home.php — landing page, kategori, pencarian, newsletter

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * Landing page dan penjelajahan katalog.
 *   /                  index()      landing page
 *   /kategori/kereta   kategori()   produk per kategori
 *   /cari?q=...        cari()       hasil pencarian
 *   /langganan         langganan()  (POST) daftar newsletter promo
 */
class Home extends MY_Controller
{
    public function index()
    {
        $this->render('home/landing', array(
            'title'     => 'ShopEase — Toko Miniatur & Koleksi',
            'unggulan'  => $this->katalog_model->by_slug('lokomotif-uap'),
            'pajangan'  => array(
                $this->katalog_model->by_slug('vespa-klasik'),
                $this->katalog_model->by_slug('kapal-pinisi'),
                $this->katalog_model->by_slug('becak-jogja'),
            ),
            'terbaru'   => $this->katalog_model->terbaru(4),
            'terlaris'  => $this->katalog_model->terlaris(5),
            'promo'     => $this->katalog_model->promo_aktif(),
            'langkah'   => $this->katalog_model->langkah_belanja(),
            'testimoni' => $this->katalog_model->testimoni(),
            'total_produk' => count($this->katalog_model->semua()),
        ));
    }

    public function kategori($slug = NULL)
    {
        $kategori = $this->katalog_model->kategori_by_slug($slug);
        if ( ! $kategori)
        {
            $this->tidak_ada();
        }

        $urut = $this->input->get('urut', TRUE);
        $produk = $this->katalog_model->by_kategori($slug);
        $produk = $this->_urutkan($produk, $urut);

        $this->render('home/daftar', array(
            'title'    => $kategori->nama.' — ShopEase',
            'judul'    => $kategori->nama,
            'subjudul' => $kategori->ket,
            'produk'   => $produk,
            'aktif'    => $slug,
            'urut'     => $urut,
        ));
    }

    public function cari()
    {
        $q = trim((string) $this->input->get('q', TRUE));
        $produk = $this->katalog_model->cari($q);

        $this->render('home/daftar', array(
            'title'    => 'Cari "'.$q.'" — ShopEase',
            'judul'    => $q === '' ? 'Cari miniatur' : 'Hasil untuk "'.$q.'"',
            'subjudul' => $q === '' ? 'Ketik nama miniatur, skala, atau bahan.' : count($produk).' miniatur ditemukan.',
            'produk'   => $produk,
            'aktif'    => NULL,
            'urut'     => NULL,
            'q'        => $q,
        ));
    }

    public function langganan()
    {
        if ($this->input->method() !== 'post')
        {
            redirect('');
        }

        $this->form_validation->set_rules('email', 'Email', 'trim|required|valid_email|max_length[100]');
        if ($this->form_validation->run())
        {
            // Belum ada database: email cukup diingat di session agar form berubah jadi pesan terima kasih.
            $this->session->set_userdata('pelanggan_newsletter', $this->input->post('email', TRUE));
            $this->session->set_flashdata('newsletter', array('ok' => TRUE, 'pesan' => 'Terdaftar. Info diskon berikutnya dikirim ke '.$this->input->post('email', TRUE).'.'));
        }
        else
        {
            $this->session->set_flashdata('newsletter', array('ok' => FALSE, 'pesan' => strip_tags(form_error('email'))));
        }
        redirect('#newsletter');
    }

    /** urut: murah | mahal | terlaris | (default) terbaru */
    private function _urutkan(array $produk, $urut)
    {
        $cmp = array(
            'murah'    => function ($a, $b) { return $a->harga_akhir - $b->harga_akhir; },
            'mahal'    => function ($a, $b) { return $b->harga_akhir - $a->harga_akhir; },
            'terlaris' => function ($a, $b) { return $b->terjual - $a->terjual; },
        );
        usort($produk, isset($cmp[$urut]) ? $cmp[$urut] : function ($a, $b) { return strcmp($b->masuk, $a->masuk); });
        return $produk;
    }
}

controllers/Keranjang.php — shopping cart di session

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * Keranjang belanja sederhana di session: array(slug => jumlah).
 *   GET  /keranjang          lihat isi
 *   POST /keranjang/tambah   tambah produk
 *   POST /keranjang/hapus    hapus satu produk
 */
class Keranjang extends MY_Controller
{
    const MAKS = 10;

    public function index()
    {
        $isi = array();
        $total = 0;
        foreach ($this->_isi() as $slug => $qty)
        {
            $p = $this->katalog_model->by_slug($slug);
            if ( ! $p) continue;
            $isi[] = (object) array('p' => $p, 'qty' => $qty, 'subtotal' => $p->harga_akhir * $qty);
            $total += $p->harga_akhir * $qty;
        }

        $this->render('keranjang/index', array(
            'title' => 'Keranjang — ShopEase',
            'isi'   => $isi,
            'total' => $total,
            'hemat' => array_sum(array_map(function ($i) { return ($i->p->harga - $i->p->harga_akhir) * $i->qty; }, $isi)),
        ));
    }

    public function tambah()
    {
        $this->_wajib_post();
        $slug = (string) $this->input->post('slug', TRUE);
        $produk = $this->katalog_model->by_slug($slug);
        if ( ! $produk)
        {
            $this->tidak_ada();
        }

        $isi = $this->_isi();
        $baru = min(self::MAKS, $produk->stok, (isset($isi[$slug]) ? $isi[$slug] : 0) + max(1, (int) $this->input->post('qty')));
        $isi[$slug] = $baru;
        $this->session->set_userdata('keranjang', $isi);
        $this->session->set_flashdata('notif', $produk->nama.' masuk keranjang ('.$baru.' pcs).');

        // kartu produk mengirim field "kembali" (URI halaman asal; string kosong = beranda).
        // Form detail produk tidak mengirimnya, sehingga diarahkan ke keranjang.
        $kembali = $this->input->post('kembali');
        redirect($kembali !== NULL && strpos($kembali, '//') === FALSE ? $kembali : 'keranjang');
    }

    public function hapus()
    {
        $this->_wajib_post();
        $isi = $this->_isi();
        unset($isi[(string) $this->input->post('slug')]);
        $this->session->set_userdata('keranjang', $isi);
        redirect('keranjang');
    }

    private function _isi()
    {
        return (array) $this->session->userdata('keranjang');
    }

    private function _wajib_post()
    {
        if ($this->input->method() !== 'post')
        {
            show_error('Metode tidak diizinkan.', 405);
        }
    }
}

controllers/Halaman.php — halaman statis & 404

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * Halaman informasi. Tiga URL (/tentang, /cara-belanja, /kontak) diarahkan ke
 * satu method tampil($slug) lewat satu baris route regex.
 * tidak_ditemukan() dipakai sebagai $route['404_override'].
 */
class Halaman extends MY_Controller
{
    public function tampil($slug = 'tentang')
    {
        $judul = array('tentang' => 'Tentang ShopEase', 'cara-belanja' => 'Cara belanja', 'kontak' => 'Hubungi kami');
        if ( ! isset($judul[$slug]))
        {
            $this->tidak_ada();
        }

        $this->render('halaman/'.str_replace('-', '_', $slug), array(
            'title'   => $judul[$slug].' — ShopEase',
            'judul'   => $judul[$slug],
            'langkah' => $this->katalog_model->langkah_belanja(),
        ));
    }

    public function tidak_ditemukan()
    {
        $this->tidak_ada();
    }
}

    /* ---------------- kategori ---------------- */

    public function kategori()
    {
        $hasil = array();
        foreach ($this->kategori as $slug => $k)
        {
            $k['slug'] = $slug;
            $k['jumlah'] = count($this->by_kategori($slug));
            $hasil[] = (object) $k;
        }
        return $hasil;
    }

    public function kategori_by_slug($slug)
    {
        return isset($this->kategori[$slug]) ? (object) ($this->kategori[$slug] + array('slug' => $slug)) : NULL;

    /* ---------------- produk ---------------- */

    /** Semua produk, sudah dilengkapi harga setelah promo */
    public function semua()
    {
        return array_map(array($this, '_lengkapi'), $this->produk);
    }

    public function by_slug($slug)
    {
        foreach ($this->semua() as $p)
        {
            if ($p->slug === $slug) return $p;
        }
        return NULL;

    public function by_kategori($slug)
    {
        return array_values(array_filter($this->semua(), function ($p) use ($slug) { return $p->kategori === $slug; }));

        $data = $this->semua();
        usort($data, function ($a, $b) { return strcmp($b->masuk, $a->masuk); });
        return array_slice($data, 0, $limit);

    public function terlaris($limit = 5)
    {
        $data = $this->semua();
        usort($data, function ($a, $b) { return $b->terjual - $a->terjual; });
        return array_slice($data, 0, $limit);

        return array_values(array_filter($this->semua(), function ($p) use ($kata) {
            return mb_strpos(mb_strtolower($p->nama.' '.$p->deskripsi.' '.$p->skala), $kata) !== FALSE;
        }));
    }

    /** Produk lain dari kategori yang sama */
    public function serupa($produk, $limit = 3)
    {
        $lain = array_filter($this->by_kategori($produk->kategori), function ($p) use ($produk) { return $p->slug !== $produk->slug; });
        return array_slice(array_values($lain), 0, $limit);
    }

    /* ---------------- promo ---------------- */

    public function promo_aktif()
    {
        $now = date('Y-m-d H:i:s');
        if ($now < $this->promo['mulai'] || $now > $this->promo['selesai']) return NULL;
        $p = (object) $this->promo;
        $p->produk = $this->by_kategori($p->kategori);
        return $p;
    }

    /* ---------------- helper internal ---------------- */

    private function _lengkapi(array $p)
    {
        $p = (object) $p;
        $p->gambar = 'assets/img/produk/'.$p->slug.'.png';
        $p->kategori_nama = $this->kategori[$p->kategori]['nama'];
        $p->diskon = 0;
        $now = date('Y-m-d H:i:s');
        if ($p->kategori === $this->promo['kategori'] && $now >= $this->promo['mulai'] && $now <= $this->promo['selesai'])
        {
            $p->diskon = $this->promo['persen'];
        }
        $p->harga_akhir = (int) round($p->harga * (100 - $p->diskon) / 100, -2);
        $p->baru = (strtotime($p->masuk) >= strtotime('-14 days'));
        return $p;
    }

Kesimpulan

Dengan memodifikasi routing, URL ShopEase menjadi ramah pengguna (/produk/kapal-pinisi, /kategori/kereta, /cara-belanja). Satu route regex bisa melayani beberapa halaman, dan 404_override memberi halaman 404 sendiri. Controller bertugas menyiapkan data, sedangkan MY_Controller mengurus data global dan layout sehingga controller lain tetap ringkas. View dipecah menjadi layout, partial, dan section, sehingga komponen seperti kartu produk dan label harga cukup ditulis sekali dan dipakai di banyak halaman. Hal ini sesuai dengan tujuan penerapan MVC pada studi kasus: aplikasi lebih mudah dipahami dan dikembangkan oleh programmer berikutnya.

Tugas 7 PBKK — CodeIgniter 3: Tokobuah, SIAKAD Presensi & MerchHima

📂 Source code lengkap: github.com/miaikoo/2026-framework-based-programming

Pada tugas ini saya membangun tiga aplikasi web dengan CodeIgniter 3.1.13, PHP, dan MySQL/MariaDB di atas XAMPP. Latihan 1 adalah aplikasi Tokobuah yang mencakup materi dasar CodeIgniter (PBKK 2-1 sampai 2-7). Latihan 2 terdiri dari dua studi kasus: SIAKAD Presensi (sistem akademik dengan presensi berbasis kode dan kelayakan UAS) dan MerchHima (toko online preorder merchandise himpunan).

Lingkungan & Persiapan

  • Windows 11, XAMPP (Apache 2.4, MariaDB 10.11, PHP 8.5), CodeIgniter 3.1.13.
  • Ketiga aplikasi memakai satu folder system/ bersama: di setiap index.php diubah $system_path = '../system';.
  • CodeIgniter 3.1.13 dirilis sebelum PHP 8.2, sehingga di PHP 8.5 muncul pesan Deprecated (misalnya Creation of dynamic property). Pesan itu hanya peringatan, jadi disembunyikan dengan error_reporting(E_ALL & ~E_DEPRECATED & ~E_USER_DEPRECATED).
  • base_url dihitung otomatis, index_page dikosongkan dan ditambah .htaccess agar URL tanpa index.php, csrf_protection diaktifkan, dan bahasa diubah ke indonesian (file form_validation_lang.php dan upload_lang.php diterjemahkan sendiri).

Cara menjalankan: letakkan folder di htdocs, start Apache & MySQL, import database/tokobuah.sql, siakad.sql, dan merch_po.sql lewat phpMyAdmin, lalu buka http://localhost/pbkk7/tokobuah/, /siakad/, dan /merch-po/.

/**
 * Semua controller di folder admin/ mewarisi class ini, jadi pengecekan login
 * cukup ditulis sekali di sini, tidak diulang di setiap controller.
 */
class Admin_Controller extends MY_Controller
{
    /** @var object data user yang sedang login */
    protected $user;

    public function __construct()
    {
        parent::__construct();

        if ( ! $this->session->userdata('logged_in'))
        {
            $this->session->set_flashdata('warning', 'Silakan login terlebih dahulu.');
            redirect('login');
        }

        $this->user = (object) array(
            'id'        => $this->session->userdata('user_id'),
            'username'  => $this->session->userdata('username'),
            'full_name' => $this->session->userdata('full_name'),
            'role'      => $this->session->userdata('role'),
        );
        // tersedia di semua view sebagai $user
        $this->load->vars(array('user' => $this->user));
    }

    /** Hanya role tertentu yang boleh melanjutkan. */
    protected function only_role($role)
    {
        if ($this->user->role !== $role)
        {
            $this->session->set_flashdata('danger', 'Akses ditolak: fitur ini khusus '.$role.'.');
            redirect('admin/products');
        }
    }
}

models/Product_model.php

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

class Product_model extends CI_Model
{
    private $table = 'products';

    /** Aturan validasi form tambah/edit produk */
    public function rules()
    {
        return array(
            array('field' => 'name',        'label' => 'Nama produk', 'rules' => 'trim|required|max_length[100]'),
            array('field' => 'category',    'label' => 'Kategori',    'rules' => 'required|in_list[Lokal,Impor]'),
            array('field' => 'price',       'label' => 'Harga',       'rules' => 'trim|required|is_natural_no_zero'),
            array('field' => 'unit',        'label' => 'Satuan',      'rules' => 'required|in_list[kg,buah,sisir,pack]'),
            array('field' => 'stock',       'label' => 'Stok',        'rules' => 'trim|required|is_natural'),
            array('field' => 'description', 'label' => 'Deskripsi',   'rules' => 'trim|max_length[500]'),
        );
    }

    /**
     * Ambil semua produk, bisa difilter kata kunci dan kategori.
     */
    public function get_all($keyword = NULL, $category = NULL, $order = 'name ASC')
    {
        if ($keyword)
        {
            $this->db->group_start()
                ->like('name', $keyword)
                ->or_like('description', $keyword)
                ->group_end();
        }
        if (in_array($category, array('Lokal', 'Impor'), TRUE))
        {
            $this->db->where('category', $category);
        }
        return $this->db->order_by($order)->get($this->table)->result();
    }

    public function get_by_id($id)
    {
        return $this->db->get_where($this->table, array('product_id' => (int) $id))->row();
    }

    public function insert(array $data)
    {
        $this->db->insert($this->table, $data);
        return $this->db->insert_id();
    }

    public function update($id, array $data)
    {
        return $this->db->update($this->table, $data, array('product_id' => (int) $id));
    }

    public function delete($id)
    {
        return $this->db->delete($this->table, array('product_id' => (int) $id));
    }

    /** Angka-angka untuk kartu ringkasan di dashboard */
    public function summary()
    {
        return $this->db->select('COUNT(*) AS total_produk,
                COALESCE(SUM(stock), 0) AS total_stok,
                COALESCE(SUM(stock * price), 0) AS nilai_stok,
                COALESCE(SUM(stock < 20), 0) AS stok_menipis', FALSE)
            ->get($this->table)->row();
    }

    public function latest($limit = 5)
    {
        return $this->db->order_by('product_id', 'DESC')->limit($limit)->get($this->table)->result();
    }
}

controllers/admin/Products.php — CRUD + upload

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * CRUD produk + upload gambar.
 *   GET  /admin/products            daftar produk
 *   GET  /admin/products/add        form tambah     (POST = simpan)
 *   GET  /admin/products/edit/3     form edit       (POST = simpan)
 *   POST /admin/products/delete     hapus (khusus role admin)
 */
class Products extends Admin_Controller
{
    const UPLOAD_DIR = './upload/product/';

    public function __construct()
    {
        parent::__construct();
        $this->load->model('product_model');
    }

    public function index()
    {
        $data = array(
            'title'    => 'Daftar Produk',
            'crumbs'   => array('Produk' => NULL),
            'products' => $this->product_model->get_all(NULL, NULL, 'product_id DESC'),
        );
        $this->load->view('admin/products/index', $data);
    }

    public function add()
    {
        $this->form_validation->set_rules($this->product_model->rules());

        if ($this->form_validation->run() === TRUE)
        {
            $data = $this->_form_data();

            $image = $this->_upload_image($data['name']);
            if ($image === FALSE)
            {
                return $this->_show_form('add', NULL);
            }
            $data['image'] = $image ?: 'default.png';

            $this->product_model->insert($data);
            $this->session->set_flashdata('success', 'Produk <b>'.html_escape($data['name']).'</b> berhasil ditambahkan.');
            redirect('admin/products');
        }

        $this->_show_form('add', NULL);
    }

    public function edit($id = NULL)
    {
        $product = $this->product_model->get_by_id($id);
        if ( ! $product)
        {
            show_404();
        }

        $this->form_validation->set_rules($this->product_model->rules());

        if ($this->form_validation->run() === TRUE)
        {
            $data = $this->_form_data();

            $image = $this->_upload_image($data['name']);
            if ($image === FALSE)
            {
                return $this->_show_form('edit', $product);
            }
            if ($image)
            {
                // gambar baru berhasil diupload, hapus gambar lama
                $this->_delete_image($product->image);
                $data['image'] = $image;
            }

            $this->product_model->update($product->product_id, $data);
            $this->session->set_flashdata('success', 'Produk <b>'.html_escape($data['name']).'</b> berhasil diperbarui.');
            redirect('admin/products');
        }

        $this->_show_form('edit', $product);
    }

    public function delete()
    {
        // hapus hanya lewat POST (form + token CSRF), bukan lewat link GET
        if ($this->input->method() !== 'post')
        {
            show_error('Metode tidak diizinkan', 405);
        }
        $this->only_role('admin');

        $product = $this->product_model->get_by_id($this->input->post('product_id'));
        if ($product)
        {
            $this->product_model->delete($product->product_id);
            $this->_delete_image($product->image);
            $this->session->set_flashdata('success', 'Produk <b>'.html_escape($product->name).'</b> sudah dihapus.');
        }
        redirect('admin/products');
    }

    /* ------------------------------------------------------------------ */

    private function _form_data()
    {
        return array(
            'name'        => $this->input->post('name', TRUE),
            'category'    => $this->input->post('category', TRUE),
            'price'       => (int) $this->input->post('price'),
            'unit'        => $this->input->post('unit', TRUE),
            'stock'       => (int) $this->input->post('stock'),
            'description' => $this->input->post('description', TRUE),
        );
    }

    private function _show_form($mode, $product)
    {
        $label = $mode === 'add' ? 'Tambah Produk' : 'Edit Produk';
        $this->load->view('admin/products/form', array(
            'title'        => $label,
            'crumbs'       => array('Produk' => 'admin/products', $label => NULL),
            'mode'         => $mode,
            'product'      => $product,
            'upload_error' => isset($this->upload) ? $this->upload->display_errors('', '') : '',
        ));
    }

    /**
     * Upload gambar produk.
     * @return string|null|false  nama file baru, NULL jika tidak ada file dipilih, FALSE jika gagal
     */
    private function _upload_image($product_name)
    {
        if (empty($_FILES['image']['name']))
        {
            return NULL;
        }

        $this->load->library('upload', array(
            'upload_path'   => self::UPLOAD_DIR,
            'allowed_types' => 'jpg|jpeg|png|webp',
            'max_size'      => 2048, // KB
            'file_name'     => url_title($product_name, '-', TRUE).'-'.time(),
            'overwrite'     => FALSE,
        ));

        return $this->upload->do_upload('image') ? $this->upload->data('file_name') : FALSE;
    }

    private function _delete_image($filename)
    {
        $path = self::UPLOAD_DIR.basename($filename);
        if ($filename && $filename !== 'default.png' && is_file($path))
        {
            unlink($path);
        }
    }
}

controllers/Auth.php — login & logout

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

class Auth extends CI_Controller
{
    public function __construct()
    {
        parent::__construct();
        $this->load->model('user_model');
    }

    public function login()
    {
        // sudah login? langsung ke dashboard
        if ($this->session->userdata('logged_in'))
        {
            redirect('admin');
        }

        $this->form_validation->set_rules('login', 'Username / email', 'trim|required');
        $this->form_validation->set_rules('password', 'Password', 'required');

        if ($this->form_validation->run() === TRUE)
        {
            $user = $this->user_model->attempt(
                $this->input->post('login', TRUE),
                $this->input->post('password')
            );

            if ($user)
            {
                // ganti session id setelah login untuk mencegah session fixation
                $this->session->sess_regenerate(TRUE);
                $this->session->set_userdata(array(
                    'logged_in' => TRUE,
                    'user_id'   => $user->user_id,
                    'username'  => $user->username,
                    'full_name' => $user->full_name,
                    'role'      => $user->role,
                ));
                $this->session->set_flashdata('success', 'Selamat datang, '.$user->full_name.'!');
                redirect('admin');
            }

            $this->session->set_flashdata('danger', 'Username/email atau password salah.');
            redirect('login');
        }

        $this->load->view('auth/login', array('title' => 'Login Admin - Tokobuah'));
    }

    public function logout()
    {
        $this->session->unset_userdata(array('logged_in', 'user_id', 'username', 'full_name', 'role'));
        $this->session->set_flashdata('success', 'Anda sudah logout.');
        redirect('login');
    }
}

Latihan 2a — SIAKAD Presensi

SIAKAD ini berfokus pada presensi kuliah. Dosen membuka sesi pertemuan dan sistem membuat kode presensi 6 karakter yang berlaku selama durasi tertentu. Mahasiswa memasukkan kode itu untuk tercatat hadir. Ketika sesi ditutup, peserta yang belum presensi otomatis tercatat Alpa. Rekap kehadiran menentukan apakah mahasiswa boleh mengikuti UAS (≥ 75%). Tampilan memakai AdminLTE 3.

Rancangan Tabel

  • users — akun login (role admin/dosen/mahasiswa), terhubung ke dosen_id atau mahasiswa_id.
  • mahasiswa, dosen, matakuliah — master data.
  • kelas — mata kuliah + dosen + jadwal (hari, jam, ruang); peserta_kelas — relasi N–N mahasiswa dan kelas.
  • pertemuan — sesi tatap muka dengan kode, status dibuka/ditutup, dan batas waktu.
  • presensi — status H/I/S/A per mahasiswa per pertemuan (UNIQUE pertemuan + mahasiswa).
  • pengumuman — untuk semua/dosen/mahasiswa.
  • VIEW v_rekap_kehadiran — jumlah H/I/S/A dan persentase kehadiran per mahasiswa per kelas.

-- View rekap kehadiran per mahasiswa per kelas.
-- Dihitung dari pertemuan yang sudah ditutup dan tercatat untuk mahasiswa tersebut
-- (saat sesi ditutup semua peserta pasti tercatat, minimal sebagai Alpa), sehingga
-- mahasiswa yang baru ditambahkan ke kelas tidak dihitung absen untuk pertemuan sebelumnya.
CREATE VIEW v_rekap_kehadiran AS
SELECT pk.kelas_id, pk.mahasiswa_id,
       COUNT(pr.id)                       AS total_pertemuan,
       COALESCE(SUM(pr.status = 'H'), 0)  AS hadir,
       COALESCE(SUM(pr.status = 'I'), 0)  AS izin,
       COALESCE(SUM(pr.status = 'S'), 0)  AS sakit,
       COALESCE(SUM(pr.status = 'A'), 0)  AS alpa,
       ROUND(COALESCE(SUM(pr.status = 'H'), 0) / NULLIF(COUNT(pr.id), 0) * 100, 1) AS persen
FROM peserta_kelas pk
LEFT JOIN pertemuan p ON p.kelas_id = pk.kelas_id AND p.status = 'ditutup'
LEFT JOIN presensi pr ON pr.pertemuan_id = p.id AND pr.mahasiswa_id = pk.mahasiswa_id
GROUP BY pk.kelas_id, pk.mahasiswa_id;

core/MY_Model.php — CRUD generik

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * Model dasar berisi operasi CRUD yang sama untuk semua tabel.
 * Model turunan cukup mengisi $table (dan $order bila perlu).
 */
class MY_Model extends CI_Model
{
    protected $table;
    protected $primary_key = 'id';
    protected $order = 'id ASC';

    public function all()
    {
        return $this->db->order_by($this->order)->get($this->table)->result();
    }

    public function find($id)
    {
        return $this->db->get_where($this->table, array($this->primary_key => $id))->row();
    }

    public function insert(array $data)
    {
        $this->db->insert($this->table, $data);
        return $this->db->insert_id();
    }

    public function update($id, array $data)
    {
        return $this->db->update($this->table, $data, array($this->primary_key => $id));
    }

    public function delete($id)
    {
        return $this->db->delete($this->table, array($this->primary_key => $id));
    }

    public function count()
    {
        return $this->db->count_all($this->table);
    }

    /** Pilihan untuk dropdown: array(id => label) */
    public function dropdown($label_field)
    {
        $options = array();
        foreach ($this->all() as $row)
        {
            $options[$row->{$this->primary_key}] = $row->$label_field;
        }
        return $options;
    }
}

models/Pertemuan_model.php — kode presensi, buka & tutup sesi

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * Sesi pertemuan kuliah + logika kode presensi.
 */
class Pertemuan_model extends MY_Model
{
    protected $table = 'pertemuan';

    /** Huruf yang mudah dibaca (tanpa O/0, I/1) */
    const ALFABET_KODE = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';

    public function by_kelas($kelas_id)
    {
        return $this->db->select("p.*,
                SUM(pr.status = 'H') AS hadir,
                COUNT(pr.id) AS tercatat", FALSE)
            ->from('pertemuan p')
            ->join('presensi pr', 'pr.pertemuan_id = p.id', 'left')
            ->where('p.kelas_id', $kelas_id)
            ->group_by('p.id')
            ->order_by('p.pertemuan_ke', 'DESC')
            ->get()->result();
    }

    /** Sesi yang sedang dibuka pada suatu kelas (maks. satu) */
    public function aktif($kelas_id)
    {
        return $this->db->where(array('kelas_id' => $kelas_id, 'status' => 'dibuka'))
            ->get($this->table)->row();
    }

    public function next_number($kelas_id)
    {
        $row = $this->db->select_max('pertemuan_ke')->get_where($this->table, array('kelas_id' => $kelas_id))->row();
        return (int) $row->pertemuan_ke + 1;
    }

    /**
     * Buka sesi baru: buat kode acak 6 karakter yang berlaku $durasi menit.
     */
    public function buka($kelas_id, $topik, $durasi)
    {
        $now = time();
        $id = $this->insert(array(
            'kelas_id'     => $kelas_id,
            'pertemuan_ke' => $this->next_number($kelas_id),
            'tanggal'      => date('Y-m-d', $now),
            'topik'        => $topik,
            'kode'         => $this->generate_kode(),
            'status'       => 'dibuka',
            'dibuka_at'    => date('Y-m-d H:i:s', $now),
            'berakhir_at'  => date('Y-m-d H:i:s', $now + $durasi * 60),
        ));
        return $this->find($id);
    }

    /**
     * Tutup sesi. Peserta yang belum presensi otomatis dicatat Alpa (A).
     * INSERT IGNORE + UNIQUE(pertemuan_id, mahasiswa_id) menjaga presensi yang sudah ada.
     */
    public function tutup($pertemuan)
    {
        $this->db->trans_start();
        $this->db->update($this->table, array(
            'status'      => 'ditutup',
            'berakhir_at' => min($pertemuan->berakhir_at, date('Y-m-d H:i:s')),
        ), array('id' => $pertemuan->id));

        $this->db->query(
            "INSERT IGNORE INTO presensi (pertemuan_id, mahasiswa_id, status)
             SELECT ?, pk.mahasiswa_id, 'A' FROM peserta_kelas pk WHERE pk.kelas_id = ?",
            array($pertemuan->id, $pertemuan->kelas_id)
        );
        $this->db->trans_complete();
    }

    /** Sesi yang melewati batas waktu ditutup otomatis (dipanggil saat halaman terkait dibuka). */
    public function tutup_kedaluwarsa()
    {
        $expired = $this->db->where('status', 'dibuka')
            ->where('berakhir_at <', date('Y-m-d H:i:s'))
            ->get($this->table)->result();
        foreach ($expired as $p)
        {
            $this->tutup($p);
        }
    }

    /** Cari sesi aktif berdasarkan kode yang diketik mahasiswa */
    public function cari_kode_aktif($kode)
    {
        return $this->db->where('kode', strtoupper(trim($kode)))
            ->where('status', 'dibuka')
            ->where('berakhir_at >=', date('Y-m-d H:i:s'))
            ->get($this->table)->row();
    }

    /** Sesi aktif dari kelas-kelas yang diikuti seorang mahasiswa */
    public function aktif_untuk_mahasiswa($mahasiswa_id)
    {
        return $this->db->select('p.*, mk.nama AS matakuliah, d.nama AS dosen, pr.status AS status_saya')
            ->from('pertemuan p')
            ->join('kelas k', 'k.id = p.kelas_id')
            ->join('matakuliah mk', 'mk.id = k.matakuliah_id')
            ->join('dosen d', 'd.id = k.dosen_id')
            ->join('peserta_kelas pk', 'pk.kelas_id = k.id AND pk.mahasiswa_id = '.(int) $mahasiswa_id)
            ->join('presensi pr', 'pr.pertemuan_id = p.id AND pr.mahasiswa_id = '.(int) $mahasiswa_id, 'left')
            ->where('p.status', 'dibuka')
            ->where('p.berakhir_at >=', date('Y-m-d H:i:s'))
            ->get()->result();
    }

    /** Semua sesi yang sedang dibuka (untuk dashboard admin) */
    public function semua_aktif()
    {
        return $this->db->select('p.*, mk.nama AS matakuliah, d.nama AS dosen, k.ruang')
            ->from('pertemuan p')
            ->join('kelas k', 'k.id = p.kelas_id')
            ->join('matakuliah mk', 'mk.id = k.matakuliah_id')
            ->join('dosen d', 'd.id = k.dosen_id')
            ->where('p.status', 'dibuka')
            ->get()->result();
    }

    private function generate_kode()
    {
        do
        {
            $kode = '';
            for ($i = 0; $i < 6; $i++)
            {
                $kode .= self::ALFABET_KODE[random_int(0, strlen(self::ALFABET_KODE) - 1)];
            }
            // pastikan tidak bentrok dengan sesi lain yang masih dibuka
            $bentrok = $this->db->where(array('kode' => $kode, 'status' => 'dibuka'))->count_all_results($this->table);
        }
        while ($bentrok);

        return $kode;
    }
}

controllers/dosen/Mengajar.php — halaman dosen

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * Halaman dosen: kelas yang diampu, buka/tutup sesi presensi, ubah status, rekap.
 *   /dosen/mengajar                 daftar kelas
 *   /dosen/mengajar/kelas/4         detail kelas + riwayat pertemuan
 *   /dosen/mengajar/buka/4          (POST) buka sesi baru
 *   /dosen/mengajar/sesi/41         kode presensi + daftar hadir
 *   /dosen/mengajar/status/41       (POST) simpan status izin/sakit/alpa
 *   /dosen/mengajar/tutup/41        (POST) tutup sesi
 *   /dosen/mengajar/rekap/4         rekap kehadiran & kelayakan UAS
 */
class Mengajar extends Dosen_Controller
{
    public function __construct()
    {
        parent::__construct();
        $this->load->model(array('kelas_model', 'pertemuan_model', 'presensi_model'));
        $this->pertemuan_model->tutup_kedaluwarsa();
    }

    public function index()
    {
        $kelas = $this->kelas_model->by_dosen($this->me->dosen_id);
        foreach ($kelas as $k)
        {
            $k->sesi_aktif = $this->pertemuan_model->aktif($k->id);
        }
        $this->render('dosen/index', array('title' => 'Kelas yang Saya Ampu', 'kelas' => $kelas));
    }

    public function kelas($id = NULL)
    {
        $kelas = $this->_kelas_saya($id);
        $this->render('dosen/kelas', array(
            'title'      => $kelas->matakuliah,
            'kelas'      => $kelas,
            'pertemuan'  => $this->pertemuan_model->by_kelas($kelas->id),
            'sesi_aktif' => $this->pertemuan_model->aktif($kelas->id),
            'berikutnya' => $this->pertemuan_model->next_number($kelas->id),
        ));
    }

    public function buka($id = NULL)
    {
        $this->require_post();
        $kelas = $this->_kelas_saya($id);

        if ($this->pertemuan_model->aktif($kelas->id))
        {
            $this->flash('warning', 'Masih ada sesi yang dibuka di kelas ini. Tutup dulu sebelum membuka sesi baru.');
            redirect('dosen/mengajar/kelas/'.$kelas->id);
        }

        $this->form_validation->set_rules('topik', 'Topik', 'trim|required|max_length[150]');
        $this->form_validation->set_rules('durasi', 'Durasi', 'required|in_list[10,15,30,60,90]');
        if ( ! $this->form_validation->run())
        {
            $this->flash('danger', validation_errors(' ', ' '));
            redirect('dosen/mengajar/kelas/'.$kelas->id);
        }

        $sesi = $this->pertemuan_model->buka($kelas->id, $this->input->post('topik', TRUE), (int) $this->input->post('durasi'));
        $this->flash('success', 'Sesi pertemuan ke-'.$sesi->pertemuan_ke.' dibuka. Tampilkan kode <b>'.$sesi->kode.'</b> kepada mahasiswa.');
        redirect('dosen/mengajar/sesi/'.$sesi->id);
    }

    public function sesi($id = NULL)
    {
        list($sesi, $kelas) = $this->_sesi_saya($id);
        $daftar = $this->presensi_model->daftar_pertemuan($sesi);

        $this->render('dosen/sesi', array(
            'title'  => 'Pertemuan ke-'.$sesi->pertemuan_ke,
            'sesi'   => $sesi,
            'kelas'  => $kelas,
            'daftar' => $daftar,
            'hadir'  => count(array_filter($daftar, function ($d) { return $d->status === 'H'; })),
        ));
    }

    public function status($id = NULL)
    {
        $this->require_post();
        list($sesi, $kelas) = $this->_sesi_saya($id);

        $status     = (array) $this->input->post('status');
        $keterangan = (array) $this->input->post('keterangan');
        $peserta    = $this->kelas_model->peserta_ids($kelas->id);
        $diubah = 0;

        foreach ($status as $mahasiswa_id => $st)
        {
            // abaikan data yang tidak valid / bukan peserta kelas ini
            if ($st === '' || ! isset(Presensi_model::STATUS[$st]) || ! in_array((int) $mahasiswa_id, $peserta, TRUE))
            {
                continue;
            }
            $ket = trim(strip_tags((string) (isset($keterangan[$mahasiswa_id]) ? $keterangan[$mahasiswa_id] : '')));
            $this->presensi_model->simpan_status($sesi->id, (int) $mahasiswa_id, $st, mb_substr($ket, 0, 150));
            $diubah++;
        }

        $this->flash('success', 'Status presensi disimpan ('.$diubah.' mahasiswa).');
        redirect('dosen/mengajar/sesi/'.$sesi->id);
    }

    public function tutup($id = NULL)
    {
        $this->require_post();
        list($sesi, $kelas) = $this->_sesi_saya($id);

        {
            $this->pertemuan_model->tutup($sesi);
            $this->flash('success', 'Sesi ditutup. Mahasiswa yang belum presensi otomatis tercatat <b>Alpa</b>.');
        }
        redirect('dosen/mengajar/sesi/'.$sesi->id);

            'title'     => 'Rekap Kehadiran',
            'kelas'     => $kelas,
            'pertemuan' => $pertemuan,
            'rekap'     => $this->presensi_model->rekap_kelas($kelas->id),
            'matriks'   => $this->presensi_model->matriks_kelas($kelas->id),
        ));
    }

    /* ---------- otorisasi: dosen hanya boleh mengakses kelasnya sendiri ---------- */

    private function _kelas_saya($id)
    {
        $kelas = $this->kelas_model->detail((int) $id);
        if ( ! $kelas || (int) $kelas->dosen_id !== (int) $this->me->dosen_id)
        {
            show_404();
        }
        return $kelas;
    }

    private function _sesi_saya($id)
    {
        $sesi = $this->pertemuan_model->find((int) $id) OR show_404();
        return array($sesi, $this->_kelas_saya($sesi->kelas_id));
    }

libraries/Keranjang_belanja.php — library keranjang buatan sendiri

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

/**
 * Library keranjang belanja buatan sendiri (disimpan di session).
 * Library Cart bawaan CI3 sudah deprecated, jadi dibuat versi sederhana:
 *   session['keranjang'] = array('3|L' => 2, '4|' => 1)  -> "produk_id|ukuran" => qty
 */
class Keranjang_belanja
{
    const KEY = 'keranjang';
    const MAKS_QTY = 10;

    /** @var CI_Controller */
    protected $CI;

    public function __construct()
    {
        $this->CI =& get_instance();
    }

    public function raw()
    {
        return (array) $this->CI->session->userdata(self::KEY);
    }

    public function tambah($produk_id, $ukuran, $qty)
    {
        $items = $this->raw();
        $key = (int) $produk_id.'|'.(string) $ukuran;
        $items[$key] = min(self::MAKS_QTY, (isset($items[$key]) ? $items[$key] : 0) + max(1, (int) $qty));
        $this->CI->session->set_userdata(self::KEY, $items);
    }

    public function ubah($key, $qty)
    {
        $items = $this->raw();
        if (isset($items[$key]))
        {
            $qty = (int) $qty;
            if ($qty <= 0) unset($items[$key]);
            else $items[$key] = min(self::MAKS_QTY, $qty);
            $this->CI->session->set_userdata(self::KEY, $items);
        }
    }

    public function hapus($key)
    {
        $this->ubah($key, 0);
    }

    public function kosongkan()
    {
        $this->CI->session->unset_userdata(self::KEY);
    }

    /** jumlah barang (untuk badge di navbar) */
    public function jumlah()
    {
        return array_sum($this->raw());
    }

    /**
     * Isi keranjang lengkap dengan data produk dari database.
     * @return array array('items' => [...], 'total' => int, 'batch_id' => int|null)
     */
    public function detail()
    {
        $this->CI->load->model('produk_model');
        $items = array();
        $total = 0;
        $batch_id = NULL;

        foreach ($this->raw() as $key => $qty)
        {
            list($produk_id, $ukuran) = explode('|', $key);
            $p = $this->CI->produk_model->find((int) $produk_id);
            if ( ! $p)
            {
                $this->hapus($key); // produk sudah dihapus admin
                continue;
            }
            $items[] = (object) array(
                'key'      => $key,
                'produk'   => $p,
                'ukuran'   => $ukuran !== '' ? $ukuran : NULL,
                'qty'      => $qty,
                'subtotal' => $p->harga * $qty,
            );
            $total += $p->harga * $qty;
            $batch_id = $batch_id ?: (int) $p->batch_id;
        }
        return array('items' => $items, 'total' => $total, 'batch_id' => $batch_id);
    }
}

            redirect('keranjang');
        }
        redirect('produk/'.$produk->id);
    }

        $this->require_post();
        foreach ((array) $this->input->post('qty') as $key => $qty)

        }
        $this->flash('success', 'Keranjang diperbarui.');
        redirect('keranjang');

        $this->keranjang->hapus($this->input->post('key'));

    public function checkout()
    {
        $this->require_post();
        if ( ! $this->me)
        {
            // setelah login kembali ke keranjang (bukan ke URL checkout yang khusus POST)
            $this->session->set_userdata('setelah_login', 'keranjang');
            $this->flash('warning', 'Masuk atau daftar dulu untuk menyelesaikan pesanan. Isi keranjang Anda tetap tersimpan.');
            redirect('login');
        }
        $this->require_login('pembeli');

        $cart = $this->keranjang->detail();
        if ( ! $cart['items'])
        {
            redirect('keranjang');
        }

        // 1) batch masih dibuka?
        $batch = $this->batch_model->find($cart['batch_id']);
        if (Batch_model::status($batch) !== 'dibuka')
        {
            $this->flash('danger', 'Maaf, '.html_escape($batch->nama).' sudah ditutup.');
            redirect('keranjang');
        }

        // 2) kuota produk terbatas masih cukup?
        $per_produk = array();
        foreach ($cart['items'] as $it)
        {
            $per_produk[$it->produk->id] = (isset($per_produk[$it->produk->id]) ? $per_produk[$it->produk->id] : 0) + $it->qty;
        }
        foreach ($per_produk as $pid => $qty)
        {
            $p = $this->produk_model->detail($pid);
            $sisa = Produk_model::sisa($p);
            if ($sisa !== NULL && $qty > $sisa)
            {
                $this->flash('danger', 'Kuota <b>'.html_escape($p->nama).'</b> tinggal '.$sisa.'. Kurangi jumlahnya.');
                redirect('keranjang');
            }
        }

        // 3) simpan pesanan
        $catatan = mb_substr(trim(strip_tags((string) $this->input->post('catatan'))), 0, 255);
        $kode = $this->pesanan_model->buat($this->me->id, $batch->id, $cart['items'], $cart['total'], $catatan);
        if ( ! $kode)
        {
            $this->flash('danger', 'Pesanan gagal disimpan, coba lagi.');
            redirect('keranjang');
        }

        $this->keranjang->kosongkan();
        $this->flash('success', 'Pesanan <b>'.$kode.'</b> dibuat. Silakan transfer lalu unggah bukti pembayaran.');
        redirect('pesanan/detail/'.$kode);
    }

<?php
defined('BASEPATH') OR exit('No direct script access allowed');

class Pesanan_model extends MY_Model
{
    protected $table = 'pesanan';
    protected $order = 'created_at DESC';

    /** status => array(label, warna badge, ikon) */
    const STATUS = array(
        'menunggu_pembayaran' => array('Menunggu pembayaran', 'warning',   'fa-wallet'),
        'menunggu_verifikasi' => array('Menunggu verifikasi', 'info',      'fa-hourglass-half'),
        'lunas'               => array('Lunas · diproduksi',  'primary',   'fa-check'),
        'ditolak'             => array('Pembayaran ditolak',  'danger',    'fa-times'),
        'siap_diambil'        => array('Siap diambil',        'success',   'fa-box-open'),
        'selesai'             => array('Selesai',             'secondary', 'fa-flag-checkered'),
        'dibatalkan'          => array('Dibatalkan',          'dark',      'fa-ban'),
    );

    /** aksi admin => array(status asal, status tujuan) */
    const TRANSISI = array(
        'verifikasi' => array('menunggu_verifikasi', 'lunas'),
        'tolak'      => array('menunggu_verifikasi', 'ditolak'),
        'siap'       => array('lunas', 'siap_diambil'),
        'selesai'    => array('siap_diambil', 'selesai'),
    );

    /** status yang dihitung sebagai pembayaran sah */
    const SAH = array('lunas', 'siap_diambil', 'selesai');

    /**
     * Buat pesanan dari isi keranjang dalam satu transaksi.
     * @return string kode pesanan
     */
    public function buat($user_id, $batch_id, array $items, $total, $catatan)
    {
        $this->db->trans_start();
        $now = date('Y-m-d H:i:s');
        $id = $this->insert(array(
            'kode'       => uniqid('tmp'),   // diganti setelah id diketahui
            'user_id'    => $user_id,
            'batch_id'   => $batch_id,
            'total'      => $total,
            'status'     => 'menunggu_pembayaran',
            'catatan'    => $catatan ?: NULL,
            'created_at' => $now,
        ));
        $kode = 'MH'.date('ymd').'-'.str_pad($id, 4, '0', STR_PAD_LEFT);
        $this->update($id, array('kode' => $kode));

        foreach ($items as $it)
        {
            $this->db->insert('pesanan_item', array(
                'pesanan_id' => $id,
                'produk_id'  => $it->produk->id,
                'ukuran'     => $it->ukuran,
                'qty'        => $it->qty,
                'harga'      => $it->produk->harga,
                'subtotal'   => $it->subtotal,
            ));
        }
        $this->db->trans_complete();
        return $this->db->trans_status() ? $kode : FALSE;
    }

    public function by_kode($kode)
    {
        return $this->db->select('o.*, u.nama AS pembeli, u.email, u.no_wa, b.nama AS batch, b.rekening, b.estimasi_ambil, b.ditutup_at')
            ->from('pesanan o')
            ->join('users u', 'u.id = o.user_id')
            ->join('batch_po b', 'b.id = o.batch_id')
            ->where('o.kode', $kode)->get()->row();
    }

    public function items($pesanan_id)
    {
        return $this->db->select('i.*, p.nama, p.gambar')
            ->from('pesanan_item i')->join('produk p', 'p.id = i.produk_id')
            ->where('i.pesanan_id', $pesanan_id)->order_by('i.id')->get()->result();
    }

    public function by_user($user_id)
    {
        return $this->db->select('o.*, b.nama AS batch, (SELECT SUM(qty) FROM pesanan_item WHERE pesanan_id = o.id) AS jumlah_barang', FALSE)
            ->from('pesanan o')->join('batch_po b', 'b.id = o.batch_id')
            ->where('o.user_id', $user_id)->order_by('o.created_at', 'DESC')->get()->result();
    }

    public function daftar_admin($status = NULL, $batch_id = NULL)
    {
        if ($status && isset(self::STATUS[$status])) $this->db->where('o.status', $status);
        if ($batch_id) $this->db->where('o.batch_id', (int) $batch_id);
        return $this->db->select('o.*, u.nama AS pembeli, u.no_wa, b.nama AS batch, (SELECT SUM(qty) FROM pesanan_item WHERE pesanan_id = o.id) AS jumlah_barang', FALSE)
            ->from('pesanan o')->join('users u', 'u.id = o.user_id')->join('batch_po b', 'b.id = o.batch_id')
            ->order_by("FIELD(o.status, 'menunggu_verifikasi') DESC, o.created_at DESC", '', FALSE)
            ->get()->result();
    }

    public function hitung_per_status()
    {
        $hasil = array_fill_keys(array_keys(self::STATUS), 0);
        foreach ($this->db->select('status, COUNT(*) AS n')->group_by('status')->get($this->table)->result() as $r)
        {
            $hasil[$r->status] = (int) $r->n;
        }
        return $hasil;
    }

    public function pendapatan_sah($batch_id = NULL)
    {
        if ($batch_id) $this->db->where('batch_id', $batch_id);
        $r = $this->db->select_sum('total')->where_in('status', self::SAH)->get($this->table)->row();
        return (int) $r->total;
    }

    /** Jumlah terjual (pesanan sah) per produk dalam satu batch, untuk grafik */
    public function terjual_per_produk($batch_id)
    {
        // item dari pesanan yang belum sah tetap ikut ter-join (LEFT JOIN), jadi hanya
        // dijumlahkan bila pesanannya cocok (o.id tidak NULL)
        return $this->db->select('p.nama, COALESCE(SUM(IF(o.id IS NULL, 0, i.qty)), 0) AS qty', FALSE)
            ->from('produk p')
            ->join('pesanan_item i', 'i.produk_id = p.id', 'left')
            ->join('pesanan o', "o.id = i.pesanan_id AND o.status IN ('lunas','siap_diambil','selesai')", 'left')
            ->where('p.batch_id', $batch_id)
            ->group_by('p.id')->order_by('p.id')->get()->result();
    }

    /**
     * Rekap produksi: jumlah per produk per ukuran dari pesanan sah.
     * @return array produk_id => (object) nama, pakai_ukuran, per_ukuran[S..XXL], total
     */
    public function rekap_produksi($batch_id)
    {
        $rows = $this->db->select('p.id, p.nama, p.pakai_ukuran, i.ukuran, SUM(i.qty) AS qty')
            ->from('pesanan_item i')
            ->join('pesanan o', 'o.id = i.pesanan_id')
            ->join('produk p', 'p.id = i.produk_id')
            ->where('o.batch_id', $batch_id)
            ->where_in('o.status', self::SAH)
            ->group_by(array('p.id', 'i.ukuran'))
            ->order_by('p.id')->get()->result();

        $rekap = array();
        foreach ($rows as $r)
        {
            if ( ! isset($rekap[$r->id]))
            {
                $rekap[$r->id] = (object) array('nama' => $r->nama, 'pakai_ukuran' => (bool) $r->pakai_ukuran,
                    'per_ukuran' => array_fill_keys(Produk_model::UKURAN, 0), 'total' => 0);
            }
            if ($r->ukuran) $rekap[$r->id]->per_ukuran[$r->ukuran] = (int) $r->qty;
            $rekap[$r->id]->total += (int) $r->qty;
        }
        return $rekap;
    }

    /** Daftar pengambilan: siapa mengambil apa (pesanan sah) */
    public function daftar_ambil($batch_id)
    {
        return $this->db->select("o.kode, o.status, u.nama, u.no_wa,
                GROUP_CONCAT(CONCAT(p.nama, IF(i.ukuran IS NULL, '', CONCAT(' (', i.ukuran, ')')), ' x', i.qty) ORDER BY i.id SEPARATOR '; ') AS barang", FALSE)
            ->from('pesanan o')->join('users u', 'u.id = o.user_id')
            ->join('pesanan_item i', 'i.pesanan_id = o.id')->join('produk p', 'p.id = i.produk_id')
            ->where('o.batch_id', $batch_id)->where_in('o.status', self::SAH)
            ->group_by('o.id')->order_by('u.nama')->get()->result();
    }
}

            'o'     => $o,
            'items' => $this->pesanan_model->items($o->id),
        ));
    }

    /** POST aksi=verifikasi|tolak|siap|selesai */
    public function ubah_status($kode = NULL)
    {
        $this->require_post();
        $o = $this->pesanan_model->by_kode((string) $kode) OR show_404();
        $aksi = $this->input->post('aksi');

        if ( ! isset(Pesanan_model::TRANSISI[$aksi]) || Pesanan_model::TRANSISI[$aksi][0] !== $o->status)
        {
            $this->flash('danger', 'Aksi tidak valid untuk status pesanan saat ini.');
            redirect('admin/pesanan/detail/'.$o->kode);
        }

        $data = array('status' => Pesanan_model::TRANSISI[$aksi][1]);
        if ($aksi === 'verifikasi')
        {
            $data['diverifikasi_at'] = date('Y-m-d H:i:s');
        }
        if ($aksi === 'tolak')
        {
            $alasan = trim(strip_tags((string) $this->input->post('alasan_tolak')));
            if ($alasan === '')
            {
                $this->flash('danger', 'Isi alasan penolakan agar pembeli tahu apa yang harus diperbaiki.');
                redirect('admin/pesanan/detail/'.$o->kode);
            }
            $data['alasan_tolak'] = mb_substr($alasan, 0, 255);
        }

        $this->pesanan_model->update($o->id, $data);
        $this->flash('success', 'Status pesanan <b>'.$o->kode.'</b> menjadi <b>'.Pesanan_model::STATUS[$data['status']][0].'</b>.');
        redirect('admin/pesanan/detail/'.$o->kode);
    }

    public function bukti($kode = NULL)
    {
        $o = $this->pesanan_model->by_kode((string) $kode) OR show_404();
        kirim_bukti('./storage/bukti/', $o->bukti_bayar);
    }
